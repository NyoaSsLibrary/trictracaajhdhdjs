-- ── WIKI IMAGE FETCH ─────────────────────────────────────────────────────────
local _LL_FANDOM_BASE = "https://stealabrainrot.fandom.com/wiki/"
local _LL_imgCache    = {}

local function _llRequestFn()
    return (syn and syn.request)
        or (http and http.request)
        or http_request
        or request
end

local function _llUrlEncode(s)
    return (s:gsub("[^%w%-_%.~]", function(c)
        return string.format("%%%02X", string.byte(c))
    end))
end

local function _llWikiCandidates(displayName)
    local clean = displayName:gsub("%[.-%]%s*", ""):gsub("^%s+", ""):gsub("%s+$", "")
    clean = clean:match("^(.-)%s*%(") or clean
    local seen, list = {}, {}
    local function add(s)
        if s and s ~= "" and not seen[s] then seen[s] = true; table.insert(list, s) end
    end
    add((clean:gsub(" ", "_")))
    add(_llUrlEncode(clean))
    add((displayName:gsub(" ", "_")))
    add(_llUrlEncode(displayName))
    return list
end

local function _llFetchBody(requestFn, url)
    for attempt = 1, 3 do
        local ok, response = pcall(function()
            return requestFn({
                Url = url, Method = "GET",
                Headers = {
                    ["User-Agent"]      = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
                    ["Accept"]          = "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
                    ["Accept-Language"] = "en-US,en;q=0.5",
                    ["Cache-Control"]   = "no-cache",
                },
                Timeout = 10,
            })
        end)
        if ok and response and response.StatusCode == 200 and response.Body and response.Body ~= "" then
            return response.Body
        end
        if ok and response and response.StatusCode and response.StatusCode ~= 200 then
            return nil
        end
        if attempt < 3 then task.wait(0.5 * attempt) end
    end
    return nil
end

local function _llFetchFandomImage(displayName)
    if _LL_imgCache[displayName] ~= nil then
        return _LL_imgCache[displayName] or nil
    end
    local ok, body = pcall(function()
        local apiUrl = "https://stealabrainrot.fandom.com/api.php"
            .. "?action=query&prop=pageimages&format=json&pithumbsize=256&titles="
            .. (displayName:gsub(" ", "_"))
        local fn = _llRequestFn()
        if fn then
            local r = fn({ Url = apiUrl, Method = "GET",
                Headers = { ["User-Agent"] = "Mozilla/5.0" }, Timeout = 8 })
            return (r and r.StatusCode == 200) and r.Body or nil
        end
        return HttpService:GetAsync(apiUrl)
    end)
    if ok and body then
        local thumb = body:match('"source":"(https://[^"]+)"')
            or body:match('"url":"(https://static%.wikia%.nocookie%.net[^"]+)"')
        if thumb then
            thumb = thumb:gsub("\\/", "/"):gsub("&amp;", "&")
            _LL_imgCache[displayName] = thumb
            return thumb
        end
    end
    local requestFn = _llRequestFn()
    if requestFn then
        for _, slug in ipairs(_llWikiCandidates(displayName)) do
            local pageBody = _llFetchBody(requestFn, _LL_FANDOM_BASE .. slug)
            if pageBody then
                local og = pageBody:match('property="og:image"%s+content="([^"]+)"')
                    or pageBody:match('content="([^"]+)"%s+property="og:image"')
                    or pageBody:match('<meta%s+property="og:image"%s+content="([^"]+)"')
                if og and og ~= "" then
                    og = og:gsub("&amp;", "&"):gsub("&quot;", '"')
                    if og:find("^https?://") then
                        _LL_imgCache[displayName] = og
                        return og
                    end
                end
                for _, attr in ipairs({ "data%-src", "src" }) do
                    local last
                    for _, ext in ipairs({ "png", "jpg", "jpeg", "webp" }) do
                        local pat = attr .. '="(https://[^"]+%.' .. ext .. '[^"]*)"'
                        for u in pageBody:gmatch(pat) do
                            if u:find("static%.wikia%.nocookie%.net")
                                and not u:find("/scale%-to%-width%-down/%d%d?$") then
                                last = u
                            end
                        end
                    end
                    if last then
                        last = last:gsub("/revision/latest", "")
                        _LL_imgCache[displayName] = last
                        return last
                    end
                end
            end
        end
    end
    _LL_imgCache[displayName] = false
    return nil
end

local function formatValue(num)
    if num>=1e15 then return string.format('%.2fQ',num/1e15)
    elseif num>=1e12 then return string.format('%.2fT',num/1e12)
    elseif num>=1e9 then return string.format('%.2fB',num/1e9)
    elseif num>=1e6 then return string.format('%.2fM',num/1e6)
    elseif num>=1e3 then return string.format('%.2fK',num/1e3)
    else return tostring(math.floor(num)) end
end

local duelBases   = {}
local duelPlayers = {}

local function IsPlayerInDuel(player)
    if not player then return false end
    return player:GetAttribute("__duels_block_steal") == true
end

local function DetectDuelBases()
    duelBases = {}; duelPlayers = {}
    local Plots = workspace:FindFirstChild("Plots")
    if not Plots then return end
    for _, plot in ipairs(Plots:GetChildren()) do
        local sign = plot:FindFirstChild("PlotSign")
        if not sign then continue end
        local textLabel = sign:FindFirstChild("SurfaceGui")
            and sign.SurfaceGui:FindFirstChild("Frame")
            and sign.SurfaceGui.Frame:FindFirstChild("TextLabel")
        local baseText = textLabel and textLabel.Text or nil
        if not baseText or baseText == "Empty Base" then continue end
        local nickname = baseText:match("^(.-)'") or baseText
        local ownerPlayer = nil
        for _, p in ipairs(Players:GetPlayers()) do
            if p.DisplayName == nickname or p.Name == nickname then ownerPlayer = p; break end
        end
        if ownerPlayer and IsPlayerInDuel(ownerPlayer) then
            duelBases[plot.Name] = {plot=plot, owner=ownerPlayer, ownerName=ownerPlayer.Name}
            duelPlayers[ownerPlayer.UserId] = true
        end
    end
end

end)

local function getPlotOwnerName(plotFolder)
    local ok, result = pcall(function()
        local sign = plotFolder:FindFirstChild("PlotSign")
        if not sign then return nil end
        local sg = sign:FindFirstChild("SurfaceGui")
        local fr = sg and sg:FindFirstChild("Frame")
        local tl = fr and fr:FindFirstChild("TextLabel")
        if not tl or tl.Text == "" then return nil end
        local nickname = tl.Text:match("^(.-)'") or tl.Text
        for _, p in ipairs(Players:GetPlayers()) do
            if p.DisplayName == nickname or p.Name == nickname then return p.Name end

end

local function isPlotInDuel(plotFolder)
    if not plotFolder then return false end
    if duelBases[plotFolder.Name] then return true end
    local ownerName = getPlotOwnerName(plotFolder)
    if not ownerName then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Name == ownerName or p.DisplayName == ownerName then
            return duelPlayers[p.UserId] == true or IsPlayerInDuel(p)
        end
    end
    return false
end

local function parseMoneyPerSec(text)
    if not text or not text:find("/s") then return 0 end
    text = text:gsub("%$",""):gsub("/s",""):gsub("%s+","")
    local mult = text:find("[Tt]") and 1e12 or text:find("[Bb]") and 1e9
              or text:find("[Mm]") and 1e6 or text:find("[Kk]") and 1e3 or 1
    text = text:gsub("[KkMmBbTt]","")
    return (tonumber(text) or 0) * mult
end

        if cur.Parent and cur.Parent.Name == "Plots" then

end

        if cur.Parent and cur.Parent.Name == "Plots" then return cur end
        cur = cur.Parent
    end

local function getMutationFromObject(parent)
    if not parent then return "None" end
    -- Try visible Mutation label
    local rawMut = (parent:FindFirstChild("Mutation")
                   and parent.Mutation.Visible and parent.Mutation.Text) or nil
    if rawMut then
        local m = rawMut:gsub("<[^>]+>", ""):gsub("%s+", " "):match("^%s*(.-)%s*$")
        if m:lower():find("yin") and m:lower():find("yang") then m = "YinYang" end
        if m ~= "" then return m end
    end
    -- Try animalData-style Mutation attribute
    local mutAttr = parent:GetAttribute("Mutation")
    if mutAttr and mutAttr ~= "" then
        return tostring(mutAttr)
    end
    return "None"
end

local function scanAllBrainrots()
    local detections = {}
    local descendants = workspace:GetDescendants()
    local batchSize = 50
    for i = 1, #descendants, batchSize do
        local batchEnd = math.min(i + batchSize - 1, #descendants)
        for j = i, batchEnd do
            local obj = descendants[j]
            local isGenLabel = obj:IsA("TextLabel") and (
                obj.Name == "Generation" or obj.Name == "MoneyPerSec" or
                obj.Name == "Income" or obj.Name == "GenLabel" or
                obj.Name == "GenerationLabel" or
                (obj.Text and obj.Text:find("/s") and obj.Text:find("%$"))
            )
            if isGenLabel then
                local moneyValue = parseMoneyPerSec(obj.Text)
                local moneyM = moneyValue / 1e6
                if moneyM >= MIN_GENERATION then
                    local parent = obj.Parent
                    local name = "Unknown"
                    local JUNK_NAMES = {AnimalOverhead=true, Base=true, Spawn=true, Model=true, Part=true, Handle=true}
                    if parent then
                        local dn = parent:FindFirstChild("DisplayName")
                        if dn and dn:IsA("TextLabel") and dn.Text ~= "" and not JUNK_NAMES[dn.Text] then
                            name = dn.Text
                        elseif parent:FindFirstChild("BrainrotName") and parent.BrainrotName.Value ~= "" then
                            name = parent.BrainrotName.Value
                        elseif parent:FindFirstChild("Title") and parent.Title:IsA("TextLabel") and parent.Title.Text ~= "" and not JUNK_NAMES[parent.Title.Text] then
                            name = parent.Title.Text
                        elseif not JUNK_NAMES[parent.Name] then
                            name = parent.Name
                        elseif parent.Parent and not JUNK_NAMES[parent.Parent.Name] and parent.Parent ~= workspace then
                            name = parent.Parent.Name
                        end
                    end
                    local mutation = getMutationFromObject(parent)
                    local owner = findOwnerForBrainrot(obj)
                    local isDuel = false
                    local plotFolder = findPlotFolderForBrainrot(obj)
                    if plotFolder then isDuel = isPlotInDuel(plotFolder) end
                    table.insert(detections, {name=name, generation=obj.Text, genM=moneyM, mutation=mutation, owner=owner, isDuel=isDuel})
                end
            end
        end
        if i % (batchSize * 2) == 0 then task.wait(0.01) end
    end
    table.sort(detections, function(a,b) return a.genM > b.genM end)
    local best = nil
    for _, d in ipairs(detections) do
        d.priority = getPriority(d.name)
        if best == nil then best = d
        elseif d.priority < best.priority then best = d
        elseif d.priority == best.priority and d.genM > best.genM then best = d
        end
    end
    return detections, best
end

local function isStuckOnLoading()
    if not game:IsLoaded() then return true end
    if not Players.LocalPlayer then return true end
    if not Players.LocalPlayer.Character then return true end
    if #workspace:GetChildren() < 5 then return true end

end

local function waitForGameLoad(maxWait)
    maxWait = maxWait or 15
    local t = tick()
    repeat
        if tick()-t > maxWait then return false end
        task.wait(0.1)
    until game:IsLoaded() and Players.LocalPlayer and Players.LocalPlayer.Character
    local t2 = tick()
    repeat
        if tick()-t2 > 5 then return false end
        task.wait(0.1)
    until #workspace:GetChildren() >= 5
    return true
end

local function hopToServer()
    local startJobId = game.JobId
    local hopStart = tick()
    print("TRIC TRAC AJ: Hopping from " .. startJobId:sub(1,8) .. "...")
    while game.JobId == startJobId do
        if tick() - hopStart > 30 then
            warn("TRIC TRAC AJ: Hop timeout, force rejoining...")
            pcall(function() TeleportService:Teleport(game.PlaceId) end)
            task.wait(2); hopStart = tick(); startJobId = game.JobId
        end
        local servers = {}
        local ok, result = pcall(function()
            return game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
        end)
        if ok and result then
            pcall(function() servers = (HttpService:JSONDecode(result) or {}).data or {} end)
        end
        local found = false
        for _, s in ipairs(servers) do
            if s.id and s.id ~= game.JobId and s.playing <= (s.maxPlayers - 2) then
                print("TRIC TRAC AJ: Teleporting (" .. s.playing .. "/" .. s.maxPlayers .. ")")
                pcall(function() TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id) end)
                found = true; break
            end
        end
        if not found then
            warn("TRIC TRAC AJ: No free servers, rejoining...")
            pcall(function() TeleportService:Teleport(game.PlaceId) end)
        end
        task.wait(2)
    end
    print("TRIC TRAC AJ: Server changed -> " .. game.JobId:sub(1,8))
end

local scanComplete = false
local hopReady = false
local currentServerId = nil
local lastSentKey = ""
local serverJoinTime = tick()

local function performScan()
    scanComplete = false; hopReady = false
    currentServerId = game.JobId
    print("TRIC TRAC AJ: Loading...")
    if not waitForGameLoad(15) then
        print("TRIC TRAC AJ: Load timeout"); scanComplete=true; hopReady=true; return
    end
    local jobWait = tick()
    while (game.JobId == "" or #game.JobId < 30) and tick()-jobWait < 10 do task.wait(0.1) end
    currentServerId = game.JobId
    if not currentServerId or #currentServerId < 30 then
        print("TRIC TRAC AJ: Invalid server"); scanComplete=true; hopReady=true; return
    end
    local t = tick()
    repeat
        task.wait(0.2)
        if isStuckOnLoading() then print("TRIC TRAC AJ: Stuck"); scanComplete=true; hopReady=true; return end
        if tick()-t > 12 then print("TRIC TRAC AJ: Plots timeout"); scanComplete=true; hopReady=true; return end
    until workspace:FindFirstChild("Plots")
    task.wait(0.5)
    print("TRIC TRAC AJ: Scanning for 35 seconds...")
    local serverSentKeys = {}
    while tick() - serverJoinTime < 35 do
        local allHits, best = scanAllBrainrots()
        if best then
            local sendKey = best.name .. "|" .. currentServerId
            if not serverSentKeys[sendKey] and sendKey ~= lastSentKey then
                serverSentKeys[sendKey] = true
                lastSentKey = sendKey
                print("TRIC TRAC AJ: HIT -> " .. best.name .. " " .. best.generation)
                sendDiscord(allHits, best, currentServerId)
                sendWebSocket(best, currentServerId)
            end
        end
        task.wait(0.1)
    end
    print("TRIC TRAC AJ: 35s done, hopping...")
    scanComplete = true; hopReady = true
end
